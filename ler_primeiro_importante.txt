CREATE DATABASE calculator_financeiro;

use calculator_financeiro;

CREATE TABLE data(
    user_id int unsigned AUTO_INCREMENT NOT NULL PRIMARY KEY,
    name varchar(64) UNIQUE NOT NULL,
    senha char(64) NOT NULL,
    email char(64) NOT NULL
);

CREATE TABLE historico(
    user_id int unsigned NOT NULL,
    tipo char(64) NOT NULL,
    valor_a_ser_investido decimal(5,2) unsigned NOT NULL,
    prazo tinyint unsigned NOT NULL,
    investimento_seguinte decimal(5,2) unsigned NOT NULL,
    percentual_crescimento float unsigned NOT NULL,
    FOREIGN KEY(user_id) REFERENCES data(user_id)
);
